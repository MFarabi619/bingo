#+BEGIN_SRC go
package main

import (
	"fmt"
	"sync"
)

func hotFunc(id int) {
	fmt.Println("hit", id)
}

func main() {
	var waitGroup sync.WaitGroup

	for i := 0; i < 4; i++ {
		waitGroup.Add(1)

		go func(id int) {
			defer waitGroup.Done()

			for j := 0; j < 100; j++ {
				hotFunc(id)
			}
		}(i)
	}

	waitGroup.Wait()
}
#+END_SRC
